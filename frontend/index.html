// Tera naya live Render Backend Link const BACKEND_URL = "https://ai-interview-evaluator-hgjg.onrender.com/api/evaluate"; async function evaluateInterview(userTopic, userAnswer) { // 1. Loading state (UI mein dikhane ke liye) console.log("Evaluating... Requesting Render Server..."); try { // 2. Backend ko request bhejna const response = await fetch(BACKEND_URL, { method: "POST", headers: { // Yahan sirf Content-Type jayega, CORS error se bachne ke liye extra headers mat daalna "Content-Type": "application/json" }, body: JSON.stringify({ topic: userTopic, transcript: userAnswer, // BACKEND KO EXACT YAHI NAAM CHAHIYE user_email: "guest" }) }); // 3. Agar server ne request reject kardi (jaise 400 ya 500 error) if (!response.ok) { throw new Error(`Server ne Error ${response.status} diya hai.`); } // 4. Backend se AI ka result nikalna const result = await response.json(); console.log("AI Result Aa Gaya:", result); // 5. Result ko apni website par show karna (yahan document.getElementById laga kar result update kar lena) const finalScore = result.evaluation.score; const finalFeedback = result.evaluation.feedback; alert(`Your Score: ${finalScore}/10 \n\nFeedback: ${finalFeedback}`); return result; } catch (error) { // 6. Agar 'Failed to fetch' ya koi aur error aata hai console.error("Fetch Error Detail:", error); alert("Server se connect nahi ho paya. Ek baar dobara try kar bhai!"); } } // Button click par isko aise call karna hai (Example) /* document.getElementById("submitBtn").addEventListener("click", () => { const topicText = document.getElementById("topicInput").value; const answerText = document.getElementById("answerInput").value; evaluateInterview(topicText, answerText); }); */
